module Warehouse.RobotLink (serveRobot) where

import qualified Data.ByteString as BS
import qualified Data.ByteString.Char8 as B8
import Network.Socket (Socket, socketToHandle)
import System.IO

-- Each frame is a decimal length, a newline, then that many bytes of status.
serveRobot :: Socket -> (BS.ByteString -> IO ()) -> IO ()
serveRobot sock handleFrame = do
  h <- socketToHandle sock ReadMode
  hSetBuffering h (BlockBuffering Nothing)
  loop h
  where
    loop h = do
      eof <- hIsEOF h
      if eof
        then hClose h
        else do
          header <- B8.hGetLine h
          case B8.readInt header of
            Just (len, _) -> do
              frame <- BS.hGet h len --@@
              handleFrame frame
              loop h
            Nothing -> hClose h

module Fleet.CompressedUpload (drainSpool) where

import qualified Codec.Compression.GZip as GZip
import qualified Data.ByteString.Lazy as BL
import System.Directory (listDirectory, removeFile)
import System.FilePath (takeExtension, (</>))

-- Trackers upload gzip batches to the spool; this worker unpacks them.
drainSpool :: FilePath -> (BL.ByteString -> IO ()) -> IO ()
drainSpool spool ingest = do
  names <- filter ((== ".gz") . takeExtension) <$> listDirectory spool
  mapM_ unpackOne names
  where
    unpackOne name = do
      compressed <- BL.readFile (spool </> name)
      ingest (GZip.decompress compressed) --@@
      removeFile (spool </> name)

module Radio.FeedSocket (readFrames) where

import Control.Exception (throwIO)
import qualified Data.ByteString as BS
import qualified Data.ByteString.Char8 as B8
import System.IO

maxFrameBytes :: Int
maxFrameBytes = 65536

-- Schedule updates arrive as "<length>\n<payload>" frames from the automation desk.
readFrames :: Handle -> (BS.ByteString -> IO ()) -> IO ()
readFrames h onFrame = go
  where
    go = do
      done <- hIsEOF h
      if done
        then pure ()
        else do
          header <- B8.hGetLine h
          len <- case B8.readInt header of
            Just (n, rest) | B8.null rest, n >= 0, n <= maxFrameBytes -> pure n
            _ -> throwIO (userError "bad frame header")
          frame <- BS.hGet h len --@@
          onFrame frame
          go

module Apiary.HiveWeights (main) where

import qualified Data.Map.Strict as Map
import Data.Maybe (mapMaybe)

splitOn :: Char -> String -> [String]
splitOn sep s = case break (== sep) s of
  (chunk, []) -> [chunk]
  (chunk, _ : rest) -> chunk : splitOn sep rest

-- Lines from the scale logger look like: 2026-05-14,HIVE-07,38.4
data Reading = Reading { hive :: String, kilos :: Double }

parseReading :: String -> Maybe Reading
parseReading line =
  case reads (last fields) of
    [(w, "")] -> Just (Reading (fields !! 1) w) --@@
    _ -> Nothing
  where
    fields = splitOn ',' line

averages :: [Reading] -> Map.Map String Double
averages rs = Map.map (\(total, n) -> total / n) sums
  where
    sums = Map.fromListWith add [(hive r, (kilos r, 1)) | r <- rs]
    add (a, n) (b, m) = (a + b, n + m)

main :: IO ()
main = do
  readings <- mapMaybe parseReading . lines <$> getContents
  mapM_ (\(h, avg) -> putStrLn (h ++ "\t" ++ show avg)) (Map.toList (averages readings))

module Radio.ScheduleLines (Slot (..), main, parseSlot) where

import Data.Text (Text)
import qualified Data.Text as T
import qualified Data.Text.IO as TIO
import Data.Time (TimeOfDay)
import Data.Time.Format (defaultTimeLocale, parseTimeM)

data Slot = Slot { startsAt :: TimeOfDay, programme :: Text, presenter :: Text }

-- A schedule line: "06:00|Breakfast Show|Dana Okafor"
parseSlot :: Text -> Either Text Slot
parseSlot line =
  case T.splitOn "|" line of --@@
    [timeText, prog, who]
      | Just t <- parseTimeM True defaultTimeLocale "%H:%M" (T.unpack timeText) ->
          Right (Slot t (T.strip prog) (T.strip who))
      | otherwise -> Left ("bad start time: " <> timeText)
    _ -> Left ("expected three fields: " <> line)

main :: IO ()
main = do
  contents <- TIO.getContents
  mapM_ report (zip [1 :: Int ..] (T.lines contents))
  where
    report (n, line) = case parseSlot line of
      Right s -> TIO.putStrLn (programme s <> " with " <> presenter s)
      Left err -> TIO.putStrLn ("line " <> T.pack (show n) <> ": " <> err)

  case (requestMethod req, pathInfo req) of

      authorised <- maybe (pure False) (validToken ops) (bearer req)

        then respond (responseLBS status401 [("WWW-Authenticate", "Bearer")] "registrar token required")
        else do
          closed <- closeLoan ops ref --@@
          respond (responseLBS (if closed then status200 else status404) [] (if closed then "loan closed" else "no such loan"))
    _ -> respond (responseLBS status404 [] "")
