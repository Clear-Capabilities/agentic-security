module CommentFeed (feedHandler) where

import Control.Monad.IO.Class (liftIO)
import Data.IORef
import qualified Data.Text as T
import Happstack.Server
import Lucid

data Comment = Comment { commentAuthor :: T.Text, commentBody :: T.Text }

feedHandler :: IORef [Comment] -> ServerPart Response
feedHandler store = msum
  [ do method POST
       decodeBody (defaultBodyPolicy "/tmp" 0 4096 1024)
       author <- T.pack <$> look "author"
       body   <- T.pack <$> look "body"
       liftIO (modifyIORef store (Comment author body :))
       seeOther ("/feed" :: String) (toResponse ())
  , do method GET
       comments <- liftIO (readIORef store)
       ok (toResponseBS "text/html" (renderBS (renderFeed comments)))
  ]

renderFeed :: [Comment] -> Html ()
renderFeed cs = ul_ (mapM_ renderComment cs)

renderComment :: Comment -> Html ()
renderComment c = li_ (b_ (toHtml (commentAuthor c)) <> ": " <> toHtml (commentBody c)) --@@

{-# LANGUAGE DataKinds, TypeOperators, OverloadedStrings #-}
module ProfileCard (CardApi, cardServer) where

import Control.Monad.IO.Class (liftIO)
import Data.Text (Text)
import qualified Data.Text as T
import Database.PostgreSQL.Simple
import Servant

data Profile = Profile { handle :: Text, bio :: Text }

instance FromRow Profile where
  fromRow = Profile <$> field <*> field

type CardApi = "members" :> Capture "handle" Text :> "card" :> Get '[PlainText] Text

escapeHtml :: Text -> Text
escapeHtml = T.concatMap esc
  where
    esc '<'  = "&lt;"
    esc '>'  = "&gt;"
    esc '&'  = "&amp;"
    esc '"'  = "&quot;"
    esc '\'' = "&#39;"
    esc c    = T.singleton c

cardServer :: Connection -> Server CardApi
cardServer conn h = do
  rows <- liftIO (query conn "SELECT handle, bio FROM profiles WHERE handle = ?" (Only h))
  case rows of
    (p:_) -> pure (renderCard p)
    []    -> throwError err404

renderCard :: Profile -> Text
renderCard p = T.concat
  [ "<div class=\"card\"><h2>", escapeHtml (handle p), "</h2>"
  , "<p>", escapeHtml (bio p), "</p></div>" --@@
  ]
