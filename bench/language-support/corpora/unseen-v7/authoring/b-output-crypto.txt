  "login" :> ReqBody '[JSON] Credentials
          :> Post '[JSON] (Headers '[Header "Set-Cookie" SetCookie] NoContent)

sessionCookie :: BS.ByteString -> SetCookie
sessionCookie token = defaultSetCookie
  { setCookieName = "sid"
  , setCookieValue = token
  , setCookiePath = Just "/"
  , setCookieHttpOnly = True
  , setCookieSecure = True
  , setCookieSameSite = Just sameSiteStrict
  , setCookieMaxAge = Just 43200
  }

loginServer :: Server LoginApi
loginServer creds = do
  found <- liftIO (lookupStaff (login creds))
  case found of
    Just s | validatePassword (staffHash s) (TE.encodeUtf8 (password creds)) -> do
      raw <- liftIO (getRandomBytes 32 :: IO BS.ByteString)
      let token = B64.encodeUnpadded raw
      liftIO (storeSession (staffId s) token)
      pure (addHeader (sessionCookie token) NoContent) --@@
    _ -> throwError err401

{-# LANGUAGE OverloadedStrings #-}
module Warehouse.Login (loginPart) where

import Control.Monad.IO.Class (liftIO)
import Crypto.BCrypt (validatePassword)
import qualified Data.ByteString.Base16 as B16
import qualified Data.ByteString.Char8 as BC
import Happstack.Server
import System.Entropy (getEntropy)

import Warehouse.Staff (Picker(..), lookupPicker, storeSession)

loginPart :: ServerPart Response
loginPart = dir "login" $ do
  method POST
  badge <- look "badge"
  pin <- look "pin"
  found <- liftIO (lookupPicker badge)
  case found of
    Just picker | validatePassword (pickerPinHash picker) (BC.pack pin) -> do
      token <- liftIO (BC.unpack . B16.encode <$> getEntropy 32)
      liftIO (storeSession (pickerId picker) token)
      addCookie Session ((mkCookie "sid" token) { secure = True, httpOnly = True }) --@@
      ok (toResponse ("signed in" :: String))
    _ -> unauthorized (toResponse ("pin not accepted" :: String))
