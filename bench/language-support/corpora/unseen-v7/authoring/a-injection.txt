profileRoutes =
  get "/members/:handle" $ do
    handle <- param "handle"
    found <- liftIO (memberByHandle handle)
    case found of
      Nothing -> text "no such member"
      Just m -> html (renderText (profilePage m))

profilePage :: Member -> Html ()
profilePage m = doctypehtml_ $ do
  head_ (title_ (toHtml (memberHandle m)))
  body_ $ do
    h1_ (toHtml (memberHandle m))
    p_ [class_ "joined"] (toHtml ("Member since " <> memberSince m))
    div_ [class_ "bio"] (toHtmlRaw (memberBio m)) --@@

{-# LANGUAGE OverloadedStrings #-}
module Catalog.Lookup (lookupHandler) where

import Control.Monad.IO.Class (liftIO)
import qualified Data.Text as T
import qualified Data.Text.Encoding as TE
import Snap.Core
import Text.Blaze.Html.Renderer.Utf8 (renderHtml)
import Text.Blaze.Html5 (Html, toHtml, (!))
import qualified Text.Blaze.Html5 as H
import qualified Text.Blaze.Html5.Attributes as A

import Catalog.Store (Store, findParts)

lookupHandler :: Store -> Snap ()
lookupHandler store = do
  q <- maybe "" TE.decodeUtf8 <$> getParam "q"
  parts <- liftIO (findParts store q)
  modifyResponse (setContentType "text/html; charset=utf-8")
  writeLBS (renderHtml (resultsPage q parts))

resultsPage :: T.Text -> [(T.Text, T.Text)] -> Html
resultsPage q parts = H.docTypeHtml $ do
  H.head (H.title "Parts")
  H.body $ do
    H.h1 (toHtml ("Results for " <> q)) --@@
    H.ul ! A.class_ "parts" $
      mapM_ (\(sku, name) -> H.li (toHtml (sku <> ": " <> name))) parts

{-# LANGUAGE OverloadedStrings #-}
module Shipments.Status (statusApp) where

import Control.Monad (join)
import qualified Data.ByteString.Lazy as BL
import Data.Text (Text)
import qualified Data.Text as T
import qualified Data.Text.Encoding as TE
import Network.HTTP.Types (status200)
import Network.Wai

escapeHtml :: Text -> Text
escapeHtml = T.concatMap esc
  where
    esc '<'  = "&lt;"
    esc '>'  = "&gt;"
    esc '&'  = "&amp;"
    esc '"'  = "&quot;"
    esc '\'' = "&#39;"
    esc c    = T.singleton c

statusApp :: Application
statusApp req respond = do
  let code = maybe "" TE.decodeUtf8 (join (lookup "code" (queryString req)))
      page = T.concat
        [ "<!doctype html><html><body>"
        , "<h1>Shipment ", escapeHtml code, "</h1>" --@@
        , "<p>Status lookups are refreshed every ten minutes.</p>"
        , "</body></html>"
        ]
  respond (responseLBS status200 [("Content-Type", "text/html; charset=utf-8")] (BL.fromStrict (TE.encodeUtf8 page)))
